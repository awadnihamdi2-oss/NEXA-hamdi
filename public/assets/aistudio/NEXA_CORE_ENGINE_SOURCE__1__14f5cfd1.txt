import json

from pathlib import Path
from datetime import datetime, timezone

# ---------------- SYSTEM CONFIGURATION ----------------

NEXA_CONFIG = {
    "name": "NEXA UNIVERSAL INTELLIGENCE",
    "version": "1.0",
    "model": "gemini-3.8-flash",
    "agents": [
        "SCIENTIST",
        "INVENTOR",
        "ENGINEER",
        "CRITIC",
        "STRATEGIST"
    ],
    "human_approval_required": True
}

NEXA_STORAGE = Path("/content/NEXA_MEMORY.json")
NEXA_HISTORY = []

# ---------------- ENVIRONMENT VALIDATION ----------------

def validate_environment():

    if "client" not in globals():
        raise RuntimeError(
            "Gemini client is missing. Run the successful API connection cell first."
        )

    if not hasattr(client, "models"):
        raise RuntimeError("Gemini client is not configured correctly.")

    return True


# ---------------- MASTER INTELLIGENCE ENGINE ----------------

def build_master_prompt(objective):

    return f"""
You are NEXA UNIVERSAL INTELLIGENCE v1.0.

Your mission is to solve complex problems through rigorous
multi-perspective reasoning, research planning, innovation
and engineering.

OBJECTIVE:
{objective}

Operate as five specialized analytical roles:

1. SCIENTIST:
Analyze evidence, facts, unknowns and research requirements.

2. INVENTOR:
Generate original approaches and competing solutions.

3. ENGINEER:
Design practical technical architecture and implementation.

4. CRITIC:
Challenge assumptions, detect weaknesses and identify risks.

5. STRATEGIST:
Compare alternatives, prioritize resources and select a strategy.

UNIFIED ENGINES:

A. MASTER BRAIN:
Understand and decompose the objective.

B. RESEARCH ENGINE:
Identify evidence needed and reliable sources to investigate.
Never claim external research was performed if it was not.

C. INVENTION ENGINE:
Generate and compare at least three approaches.

D. ENGINEERING ENGINE:
Design a practical implementation architecture.

E. AGENT COUNCIL:
Present the five specialized perspectives and their disagreements.

F. VERIFICATION ENGINE:
Define measurable tests, failure conditions and quality criteria.

G. STRATEGY ENGINE:
Choose the strongest approach and justify the decision.

H. AUTONOMOUS OPERATIONS:
Produce an ordered execution plan.
Do not perform external actions or spend money.
Human approval is mandatory.

REQUIRED OUTPUT — ARABIC:

1. فهم الهدف
2. تحليل علمي
3. خطة البحث والأدلة المطلوبة
4. ثلاثة حلول مبتكرة
5. التصميم الهندسي
6. تقييم الوكلاء الخمسة
7. النقد والمخاطر
8. القرار الاستراتيجي
9. خطة التنفيذ
10. اختبارات التحقق
11. الموارد والتكاليف والقيود
12. النتيجة النهائية

QUALITY RULES:

- Be specific, logical and realistic.
- Distinguish verified facts from assumptions.
- Never invent sources, statistics or test results.
- Explain technical limitations.
- Prefer solutions that minimize cost without sacrificing quality.
- Identify missing information.
- Challenge your own conclusions.
- Do not claim to have executed code or external operations.
- Avoid generic recommendations.
"""


# ---------------- KNOWLEDGE VAULT ----------------

def save_nexa_memory(record):

    NEXA_HISTORY.append(record)

    existing = []

    if NEXA_STORAGE.exists():
        try:
            existing = json.loads(
                NEXA_STORAGE.read_text(encoding="utf-8")
            )
        except Exception:
            existing = []

    existing.append(record)

    NEXA_STORAGE.write_text(
        json.dumps(existing, ensure_ascii=False, indent=2),
        encoding="utf-8"
    )

    return len(existing)


# ---------------- VERIFICATION ENGINE ----------------

def verify_nexa_output(result):

    checks = {
        "objective_present": bool(result.get("objective")),
        "ai_response_present": bool(result.get("analysis")),
        "response_has_content": (
            len(result.get("analysis", "").strip()) > 500
        ),
        "human_approval_protected": (
            result.get("human_approval_required") is True
        )
    }

    return {
        "passed": all(checks.values()),
        "checks": checks
    }


# ---------------- UNIFIED EXECUTION ENGINE ----------------

def NEXA_RUN(objective):

    validate_environment()

    if not isinstance(objective, str) or len(objective.strip()) < 10:
        raise ValueError("Please enter a meaningful objective.")

    objective = objective.strip()

    task_id = str(uuid.uuid4())[:12]

    print("\n🧠 NEXA MASTER CORE ACTIVATED")
    print("Task:", task_id)
    print("Analyzing all intelligence modules...")

    prompt = build_master_prompt(objective)

    # One real AI request coordinates all five roles.
    response = client.models.generate_content(
        model=NEXA_CONFIG["model"],
        contents=prompt
    )

    analysis = response.text

    if not analysis or not analysis.strip():
        raise RuntimeError("The AI returned an empty response.")

    result = {
        "system": NEXA_CONFIG["name"],
        "version": NEXA_CONFIG["version"],
        "task_id": task_id,
        "timestamp": datetime.now(timezone.utc).isoformat(),
        "objective": objective,
        "active_roles": NEXA_CONFIG["agents"],
        "analysis": analysis,
        "human_approval_required": True
    }

    result["verification"] = verify_nexa_output(result)

    saved_count = save_nexa_memory(result)

    print("\n" + "=" * 60)
    print("NEXA UNIFIED INTELLIGENCE REPORT")
    print("=" * 60)

    print(result["analysis"])

    print("\n" + "=" * 60)
    print("SYSTEM VERIFICATION")
    print(json.dumps(
        result["verification"],
        ensure_ascii=False,
        indent=2
    ))

    print("Saved reports:", saved_count)
    print("Memory file:", str(NEXA_STORAGE))
    print("=" * 60)

    return result


# ---------------- AUTOMATIC SYSTEM SELF-TEST ----------------

def NEXA_SELF_TEST():

    tests = {
        "configuration": bool(NEXA_CONFIG["agents"]),
        "storage_path": NEXA_STORAGE.parent.exists(),
        "verification_engine": callable(verify_nexa_output),
        "master_prompt": callable(build_master_prompt),
        "execution_engine": callable(NEXA_RUN),
        "gemini_connection": validate_environment()
    }

    print("\nNEXA SYSTEM SELF-TEST")
    print(json.dumps(tests, indent=2, ensure_ascii=False))

    if not all(tests.values()):
        raise RuntimeError("NEXA self-test failed.")

    print("\nALL LOCAL SYSTEM TESTS PASSED")
    return True


# ---------------- START NEXA ----------------

NEXA_SELF_TEST()

MISSION = (
    "Design a technically realistic and powerful universal "
    "multi-agent intelligence system that can research, "
    "discover opportunities, invent and engineer solutions "
    "with zero initial budget."
)

NEXA_FINAL_RESULT = NEXA_RUN(MISSION)

print("\n🚀 NEXA v1.0 EXECUTION COMPLETED")

# NEXA TASK ORCHESTRATOR v0.2
# Task planning + verification + memory

from datetime import datetime, timezone
import uuid
import json

NEXA_MEMORY = []

def create_task(objective):
    if not isinstance(objective, str) or not objective.strip():
        raise ValueError("Task objective cannot be empty")

    task = {
        "id": str(uuid.uuid4())[:8],
        "objective": objective.strip(),
        "status": "PLANNED",
        "created_at": datetime.now(timezone.utc).isoformat(),
        "steps": [
            {"name": "RESEARCH", "status": "PENDING"},
            {"name": "ANALYSIS", "status": "PENDING"},
            {"name": "DESIGN", "status": "PENDING"},
            {"name": "ENGINEERING", "status": "PENDING"},
            {"name": "TESTING", "status": "PENDING"},
            {"name": "FINAL_REPORT", "status": "PENDING"}
        ],
        "approval_required": True
    }

    return task


def verify_task(task):
    checks = {
        "has_objective": bool(task.get("objective")),
        "has_steps": len(task.get("steps", [])) > 0,
        "approval_protected": task.get("approval_required") is True
    }

    return {
        "passed": all(checks.values()),
        "checks": checks
    }


def launch_nexa(objective):
    task = create_task(objective)
    verification = verify_task(task)

    if verification["passed"]:
        NEXA_MEMORY.append(task)

    print("NEXA TASK CREATED")
    print(json.dumps(task, indent=2, ensure_ascii=False))
    print("\nVERIFICATION:")
    print(json.dumps(verification, indent=2))

    return task


# First test

    "Design and build a powerful intelligent AI system"
)

print("\nNEXA ORCHESTRATOR: ONLINE")

class SearchParser(HTMLParser):
    def __init__(self):
        super().__init__()
        self.items = []
        self.current = None

    def handle_starttag(self, tag, attrs):
        attrs = dict(attrs)
        if tag == "a" and (
            "result__a" in attrs.get("class", "") or
            "result-link" in attrs.get("class", "")
        ):
            self.current = {
                "url": attrs.get("href", ""),
                "title": ""
            }

    def handle_data(self, data):
        if self.current is not None:
            self.current["title"] += data

    def handle_endtag(self, tag):
        if tag == "a" and self.current is not None:
            if self.current["title"].strip():
                self.items.append(self.current)
            self.current = None

def NEXA_WEB_SEARCH(query, max_results=6):

def NEXA_WEB_SEARCH(query, max_results=5):
    import urllib.request
    import urllib.parse
    from html.parser import HTMLParser

    class Parser(HTMLParser):
        def __init__(self):
            super().__init__()
            self.results = []
            self.current = None

        def handle_starttag(self, tag, attrs):
            a = dict(attrs)
            if tag == "a" and "result__a" in a.get("class", ""):
                self.current = {"title": "", "url": a.get("href", "")}

        def handle_data(self, data):
            if self.current is not None:
                self.current["title"] += data

        def handle_endtag(self, tag):
            if tag == "a" and self.current:
                self.results.append(self.current)
                self.current = None

    try:
        url = "https://html.duckduckgo.com/html/?q=" + urllib.parse.quote(query)
        req = urllib.request.Request(

def NEXA_ARXIV_SEARCH(topic, limit=5):
    try:
        params = urllib.parse.urlencode({
            "search_query": "all:" + topic,
            "start": 0,
            "max_results": limit,
            "sortBy": "relevance"

            url, headers={"User-Agent": "NEXA-Research/3.0"}
        )
        with urllib.request.urlopen(request, timeout=20) as response:
            xml_data = response.read()

        root = ET.fromstring(xml_data)
        ns = {"a": "http://www.w3.org/2005/Atom"}
        papers = []

                "summary": (entry.findtext("a:summary", default="", namespaces=ns)).strip(),
                "url": entry.findtext("a:id", default="", namespaces=ns),
                "published": entry.findtext("a:published", default="", namespaces=ns)
            })

        return {"available": True, "papers": papers}

    except Exception as e:
        return {"available": False, "error": str(e), "papers": []}
